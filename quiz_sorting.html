def selectionSort(listData):
    print('Algoritma Selection Sort konvensional')
    print('Data Awal =', listData)
    
    for outIter in range(len(listData)-1):
        minIndex = outIter
        
        for i in range(outIter+1, len(listData)):
            if listData[i] < listData[minIndex]:
                minIndex = i
        
        # Tukar data
        listData[outIter], listData[minIndex] = listData[minIndex], listData[outIter]
        
        print('Iterasi ke-', outIter+1, ':', listData)
        
        # 🔥 CEK apakah sudah terurut
        is_sorted = True
        for j in range(len(listData)-1):
            if listData[j] > listData[j+1]:
                is_sorted = False
                break
        
        if is_sorted:
            print('Data sudah terurut, iterasi dihentikan.')
            break

    print('Data Urut =', listData)

for k in range(len(lst)):
    tukar = False
    
    for i in range(len(lst)-1, 0, -1):
        if lst[i] < lst[i-1]:
            lst[i], lst[i-1] = lst[i-1], lst[i]
            tukar = True
    
    print("Iterasi", k+1, ":", lst)
    
    if not tukar:   # tidak ada pertukaran
        print("Sudah terurut, berhenti di iterasi", k+1)
        break

print("Hasil:", lst)
